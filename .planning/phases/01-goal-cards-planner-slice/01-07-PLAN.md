pub async fn submit_goal(&self, text: &str) -> anyhow::Result<SubmitGoalResponse> {
    let payload = serde_json::json!({ "title": text });
    let resp = self.http
        .post(format!("{}/api/board/goals", self.base_url))
        .json(&payload)
        .send().await?;
    let status = resp.status();
    let body: serde_json::Value = resp.json().await?;
    if !status.is_success() {
        anyhow::bail!("daemon returned {status}: {body}");
    }
    serde_json::from_value(body).context("parse goal response")
}

}

Overlay::Goal(ref mut form) => match key.code {
