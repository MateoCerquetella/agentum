```rust
//! End-to-end integration test for Phase 1 (goal-cards-planner-slice).
//! Exercises the full happy path through the in-process axum router:
//! goal submit -> simulated children -> dependency link -> watchdog
//! recomputes goal status -> bus events fan out.
//!
//! No tmux fixture needed: the planner-spawn branch fails (no `claude`
//! binary available in CI), the goal card is retained per CONTEXT D-07,
//! and a `goal.planner.spawn_failed` event fires — the test asserts
//! this branch works as designed.

use std::time::Duration;

use agentum_core::{Event, Status};
use agentum_server::{AppState, router};
use serde_json::json;
use tempfile::tempdir;
use tokio::sync::broadcast;
use tokio::time::timeout;
use tower::ServiceExt;
use axum::body::Body;
use axum::http::{Request, StatusCode};

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn goal_cards_full_happy_path() -> anyhow::Result<()> {
    // 1. Bootstrap: tempdir Store, AppState, router, reconciler.
    let dir = tempdir()?;
    // ... follow the existing test-bootstrap pattern from routes/board.rs::tests
    //     to construct an AppState with a fresh DB pool. Replace the env
    //     XDG_DATA_HOME if the existing pattern does this.
    let state = bootstrap_test_state(dir.path()).await?;
    let app = router().with_state(state.clone());

    // Spawn the goal reconciler manually (serve() owns this in production).
    {
        let store = state.store.clone();
        let bus = state.bus.clone();
        tokio::spawn(async move {
            agentum_watchdog::run_goal_reconciler(store, bus).await;
        });
    }

    // Subscribe BEFORE any HTTP call so no events are missed.
    let mut bus_rx = state.bus.subscribe();

    // 2. POST /api/board/goals.
    let resp = app.clone()
        .oneshot(
            Request::post("/api/board/goals")
                .header("authorization", format!("Bearer {}", state.test_token))
                .header("content-type", "application/json")
                .body(Body::from(serde_json::to_vec(&json!({
                    "title": "deliver feature",
                    "workdir": "/tmp",  // must exist so the spawn block doesn't BadRequest early
                }))?)).unwrap()
        )
        .await?;
    assert_eq!(resp.status(), StatusCode::CREATED);
    let body: serde_json::Value = read_json(resp).await?;
    let goal_id = body["goal"]["id"].as_i64().unwrap();
    let goal_key = body["goal"]["key"].as_str().unwrap().to_string();
    assert_eq!(body["goal"]["lbl"], "goal");
    assert_eq!(body["goal"]["status"], "todo");

    // 3. Observe goal.created + goal.planner.spawn_failed (no tmux in test).
    let goal_created = expect_event(&mut bus_rx, "goal.created", 500).await?;
    assert_eq!(goal_created.payload["id"], goal_id);
    let spawn_failed = expect_event(&mut bus_rx, "goal.planner.spawn_failed", 1000).await?;
    assert_eq!(spawn_failed.payload["goal_id"], goal_id);

    // 4. Simulate the planner's output by POSTing 3 children + 1 link.
    let mut child_ids = vec![];
    for key in ["a", "b", "c"] {
        let resp = app.clone()
            .oneshot(post_json(&state, "/api/board", json!({
                "title": format!("Step {}", key),
                "body": format!("key: {}\n\nDo the {} part.", key, key),
                "lbl": "feat",
                "status": "todo",
                "parent_goal_id": goal_id,
            })))
            .await?;
        assert_eq!(resp.status(), StatusCode::CREATED);
        let body: serde_json::Value = read_json(resp).await?;
        child_ids.push(body["id"].as_i64().unwrap());
    }

    // 5. Add a blocks link b -> a (b depends on a).
    let resp = app.clone()
        .oneshot(post_json(&state, "/api/board/links", json!({
            "parent_goal_id": goal_id,
            "from_key": "b",
            "to_key": "a",
            "kind": "blocks",
        })))
        .await?;
    assert_eq!(resp.status(), StatusCode::CREATED);

    // 6. Observe goal.planner.first_child fires (D-07 auto-stop pathway).
    expect_event(&mut bus_rx, "goal.planner.first_child", 1000).await?;

    // 7. PATCH child_a to doing -> goal should flip to doing.
    let resp = app.clone()
        .oneshot(patch_json(&state, &format!("/api/board/{}", child_ids[0]), json!({"status":"doing"})))
        .await?;
    assert_eq!(resp.status(), StatusCode::OK);
    let event = expect_event(&mut bus_rx, "goal.status.changed", 1000).await?;
    assert_eq!(event.payload["goal_id"], goal_id);
    assert_eq!(event.payload["to"], "doing");

    // 8. Move all children to doing then done one by one.
    for child_id in &child_ids[1..] {
        let resp = app.clone().oneshot(patch_json(&state, &format!("/api/board/{}", child_id), json!({"status":"doing"}))).await?;
        assert_eq!(resp.status(), StatusCode::OK);
    }
    // Goal stays at "doing" — no event between todo->doing transitions of additional children.

    for child_id in &child_ids[..2] {
        let resp = app.clone().oneshot(patch_json(&state, &format!("/api/board/{}", child_id), json!({"status":"done"}))).await?;
        assert_eq!(resp.status(), StatusCode::OK);
    }
    // First two done, last still doing — no goal change yet.

    // Last child to done -> goal becomes done.
    let resp = app.clone().oneshot(patch_json(&state, &format!("/api/board/{}", child_ids[2]), json!({"status":"done"}))).await?;
    assert_eq!(resp.status(), StatusCode::OK);
    let event = expect_event(&mut bus_rx, "goal.status.changed", 1000).await?;
    assert_eq!(event.payload["to"], "done");

    // 9. Reverse symmetry: move one child back to doing -> goal drops to doing.
    let resp = app.clone().oneshot(patch_json(&state, &format!("/api/board/{}", child_ids[2]), json!({"status":"doing"}))).await?;
    assert_eq!(resp.status(), StatusCode::OK);
    let event = expect_event(&mut bus_rx, "goal.status.changed", 1000).await?;
    assert_eq!(event.payload["from"], "done");
    assert_eq!(event.payload["to"], "doing");

    Ok(())
}

async fn bootstrap_test_state(dir: &std::path::Path) -> anyhow::Result<TestState> {
    // Mirror routes/board.rs:408-429 `fresh_state()` exactly. This is the
    // canonical in-process test harness for AppState. `no_auth: true` skips
    // the require_token middleware so the test does not need to mint a real
    // bearer token. The bus is the same broadcast::Sender the AppState
    // carries — handler-emitted events fan out to the test's subscription.
    use agentum_store::Store;
    use std::sync::Arc;
    use tokio::sync::broadcast;
    use agentum_server::TranscriptStore;

    let db_path = dir.join("test.sqlite");
    let store = Store::open(&db_path).await?;
    let (bus, _) = broadcast::channel(1024);
    let state = AppState {
        store: Arc::new(store),
        bus: bus.clone(),
        started_at: std::time::Instant::now(),
        version: "test",
        auth_limiter: Arc::new(agentum_server::ratelimit::RateLimiter::new(
            8,
            std::time::Duration::from_secs(60),
        )),
        cert_fingerprint: Arc::new(String::new()),
        transcripts: TranscriptStore::new(broadcast::channel(16).0),
        stream_positions: Arc::new(std::sync::Mutex::new(std::collections::HashMap::new())),
        hostname: "test".to_string(),
        no_auth: true,
    };
    // `test_token` is unused when `no_auth: true` is set, but we keep the
    // field so the helpers (`post_json` etc.) compile unchanged. Any value
    // works because the middleware is bypassed.
    Ok(TestState {
        store: state.store.clone(),
        bus,
        test_token: String::new(),
        state,
    })
}

async fn expect_event(rx: &mut broadcast::Receiver<Event>, kind: &str, ms: u64) -> anyhow::Result<Event> {
    let deadline = tokio::time::sleep(Duration::from_millis(ms));
    tokio::pin!(deadline);
    loop {
        tokio::select! {
            ev = rx.recv() => {
                let ev = ev?;
                if ev.kind == kind {
                    return Ok(ev);
                }
                // Else: drop and keep waiting.
            }
            _ = &mut deadline => {
                anyhow::bail!("timed out waiting for event {kind}");
            }
        }
    }
}

fn post_json(state: &TestState, path: &str, body: serde_json::Value) -> Request<Body> {
    Request::post(path)
        .header("authorization", format!("Bearer {}", state.test_token))
        .header("content-type", "application/json")
        .body(Body::from(serde_json::to_vec(&body).unwrap()))
        .unwrap()
}

fn patch_json(state: &TestState, path: &str, body: serde_json::Value) -> Request<Body> {
    Request::patch(path)
        .header("authorization", format!("Bearer {}", state.test_token))
        .header("content-type", "application/json")
        .body(Body::from(serde_json::to_vec(&body).unwrap()))
        .unwrap()
}

async fn read_json(resp: axum::response::Response) -> anyhow::Result<serde_json::Value> {
    let bytes = axum::body::to_bytes(resp.into_body(), 1024 * 1024).await?;
    Ok(serde_json::from_slice(&bytes)?)
}

}
```
